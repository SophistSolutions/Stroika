## Description

typename Mapping_LinkedList<KEY_TYPE, VALUE_TYPE, TRAITS>::Rep_::SharedPtrIRep  Mapping_LinkedList<KEY_TYPE, VALUE_TYPE, TRAITS>::Rep::CloneEmpty (IteratorOwnerID forIterableEnvelope) const
{
if (fData_.HasActiveIterators ()) {
CONTAINER_LOCK_HELPER_START (fData_.fLockSupport) {
// const cast because though cloning LOGICALLY makes no changes in reality we have to patch iterator lists
auto r = Iterable<KeyValuePair<KEY_TYPE, VALUE_TYPE>>::template MakeSharedPtr<Rep_> (const_cast<Rep_*> (this), forIterableEnvelope);
r->fData_.RemoveAll ();
return r;
}
CONTAINER_LOCK_HELPER_END ();
}
else {
return Iterable<KeyValuePair<KEY_TYPE, VALUE_TYPE>>::template MakeSharedPtr<Rep_> ();
}
