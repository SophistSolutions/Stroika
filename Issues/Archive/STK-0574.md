void depleteSendBuffer(int fd) 
{
#ifdef _linux_
int lastOutstanding=-1;
for(; {
int outstanding;
ioctl(fd, SIOCOUTQ, &outstanding);
if(outstanding != lastOutstanding) 
printf("Outstanding: %d\n", outstanding);
lastOutstanding = outstanding;
if(!outstanding)
break;
usleep(1000);
}
#endif
