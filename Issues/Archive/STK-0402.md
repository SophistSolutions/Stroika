## Description

This is like list reader but you start a level down.
List reader works OK when you ahve top level with repeating elt underneath and nothing else.
But if you have a struct with one or two feilds and then a repeating eltl like below, this is needed:
class   propsetReader_ : public ObjectReaderRegistry::IElementConsumer {
public:
propsetReader_ (KeyValuePair<String, ValType>* vp)
: fValuePtr_ (vp)
{
}
virtual shared_ptr<IElementConsumer>    HandleChildStart (ObjectReaderRegistry::Context& r, const Name& name) override
{
if (name == Name { L"name" }) {
return r.GetObjectReaderRegistry ().MakeContextReader (&fValuePtr_->fKey);
}
else if (name == Name { L"val" }) {
return make_shared<ObjectReaderRegistry::IgnoreNodeReader> ();
// need special reader
//return r.GetObjectReaderRegistry ().MakeContextReader (&fValuePtr_->name);
}
else {
return make_shared<ObjectReaderRegistry::IgnoreNodeReader> ();
}
}
virtual void   Deactivating (ObjectReaderRegistry::Context& r) override
{
}
private:
KeyValuePair<String, ValType>*          fValuePtr_;
};
class   propsetsMappingReader_: public ObjectReaderRegistry::IElementConsumer {
public:
propsetsMappingReader_ (Mapping<String, ValType>* v)
: fValuePtr_ (v)
{
}
virtual shared_ptr<IElementConsumer>    HandleChildStart (ObjectReaderRegistry::Context& r, const Name& name) override
{
AssertNotNull (fSubReader_);
return fSubReader_->HandleChildStart (r, name);
}
virtual  void    Activated (ObjectReaderRegistry::Context& r)
{
Assert (fSubReader_ == nullptr);
fSubReader_ = make_shared<propsetReader_> (&fCurTReading_);
}
virtual void                            Deactivating (ObjectReaderRegistry::Context& r) override
{
AssertNotNull (fSubReader_);
fSubReader_->Deactivating (r);
fValuePtr_->Add (fCurTReading_);
}
private:
shared_ptr<IElementConsumer>    fSubReader_;
KeyValuePair<String, ValType>   fCurTReading_;
Mapping<String, ValType>*       fValuePtr_;
};
