/*
  - Defining a list of callbacks assocaited with some other objects is common practice.
  - Implementers have to create a list of the shared objects, and provide an API to add
  - and remove them. This utility class is meant to just wrap that code - so its
  - shared implemenation. Its just a private re-usable utlity.
  - It also works if you want to call the callbacks direactly and synchonously, or
  - via some kind of Q like a threadpool.
 */
template	<typename Args...>
class	CallbackRegistry {
public:
     class Connection;
public:
	// Add the given callback to our list, so subsequent calls to Call()
	// will forward the message to all registered callbacks
	Connection AddCallback (std::function<void(Args)> cb);
public:
	void RemoveCallback (Connection c);
	
public:
// sends the args to each callback. if any callback throws, the remaining ones
// are not notified, and the exception is propagated
void	Call (Args ...);
public:
// example usage
// 	ThreadPool tp;
//	cbRegistry.CallViaDispatcher ([tp] (IRunnablePtr r) { to.AddTask (r);}, args);
//
template	<typename ENQUE_RUNNABLE_PROC>
