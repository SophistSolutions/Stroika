namespace {
bool    UpgradeObjectElement_ (VariantValue* variantObject, const String& eltName, const function<bool(VariantValue*)>& eltUpgrader)
{
if (variantObject->GetType () == VariantValue::Type::eMap) {
Mapping<String,VariantValue> vm = variantObject->As<Mapping<String,VariantValue>> ();
auto o = vm.Lookup (eltName);
if (o.IsPresent ()) {
VariantValue    v   =   *o;
if (eltUpgrader (&v)) {
vm.Add (eltName, v);
*variantObject = vm;
return true;
}
}
}
return false;
}
}

function<bool(VariantValue*)>    MakeStringMapUpgrader_ (const Mapping<String,String>& m) {
return [m] (VariantValue* variantValueToUpgrade) -> bool {
RequireNotNull (variantValueToUpgrade);
if (variantValueToUpgrade->GetType () == VariantValue::Type::eString) {
auto    o = m.Lookup (variantValueToUpgrade->As<String> ());
if (o.IsPresent ()) {
*variantValueToUpgrade = *o;
return true;
}
}
return false;
};
}
